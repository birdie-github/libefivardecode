cmake_minimum_required(VERSION 3.16)
project(libefivardecode VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JANSSON REQUIRED IMPORTED_TARGET jansson>=2.14)
option(EVD_RHBOOT "Format selected fixed-size device path nodes with rhboot" OFF)
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
# Reconfigure when shipped rules are edited, added, or removed.
file(GLOB registry_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/registry/*.json")
set(registry_entries "")
foreach(path IN LISTS registry_files)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  get_filename_component(name "${path}" NAME)
  file(READ "${path}" contents)
  string(FIND "${contents}" ")EVD_JSON\"" delimiter)
  if(NOT delimiter EQUAL -1)
    message(FATAL_ERROR "Raw string delimiter collision in ${name}")
  endif()
  string(APPEND registry_entries "{\"builtin:${name}\", R\"EVD_JSON(${contents})EVD_JSON\"},\n")
endforeach()
configure_file(src/builtin.inc.in builtin.inc @ONLY)

add_library(efivardecode_lib src/api.cpp src/registry.cpp src/compound.cpp)
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(efivardecode_lib PUBLIC EVD_STATIC)
  set(EVD_PC_CFLAGS "-DEVD_STATIC")
endif()
set(EVD_PC_REQUIRES "jansson >= 2.14")
set_target_properties(efivardecode_lib PROPERTIES OUTPUT_NAME efivardecode VERSION ${PROJECT_VERSION}
  SOVERSION 0 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(efivardecode_lib PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(efivardecode_lib PRIVATE cxx_std_17)
target_include_directories(efivardecode_lib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(efivardecode_lib PRIVATE EVD_BUILD EVD_DATADIR="${CMAKE_INSTALL_FULL_DATADIR}/libefivardecode/registry.d")
target_link_libraries(efivardecode_lib PRIVATE PkgConfig::JANSSON)
if(EVD_RHBOOT)
  string(APPEND EVD_PC_REQUIRES ", efivar")
  target_sources(efivardecode_lib PRIVATE src/rhboot.c)
  set_target_properties(efivardecode_lib PROPERTIES C_VISIBILITY_PRESET hidden)
  pkg_check_modules(EFIVAR REQUIRED IMPORTED_TARGET efivar)
  target_compile_definitions(efivardecode_lib PRIVATE EVD_RHBOOT)
  target_link_libraries(efivardecode_lib PRIVATE PkgConfig::EFIVAR)
endif()
add_executable(efivardecode src/cli.cpp)
target_compile_features(efivardecode PRIVATE cxx_std_17)
target_link_libraries(efivardecode PRIVATE efivardecode_lib PkgConfig::JANSSON)
install(TARGETS efivardecode_lib efivardecode)
install(FILES include/efivardecode.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY registry/ DESTINATION ${CMAKE_INSTALL_DATADIR}/libefivardecode/registry.d)
configure_file(libefivardecode.pc.in libefivardecode.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libefivardecode.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
if(BUILD_TESTING)
  add_executable(api_test tests/api.c)
  target_link_libraries(api_test PRIVATE efivardecode_lib)
  add_test(NAME api COMMAND api_test ${CMAKE_CURRENT_SOURCE_DIR}/registry)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py
      $<TARGET_FILE:efivardecode> ${CMAKE_CURRENT_SOURCE_DIR}/registry)
  endif()
endif()

option(EVD_FUZZ "Instrument the decoder and build a Clang libFuzzer target" OFF)
if(EVD_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "EVD_FUZZ requires Clang with libFuzzer")
  endif()
  target_compile_options(efivardecode_lib PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  target_link_options(efivardecode_lib PRIVATE -fsanitize=address,undefined)
  target_link_options(efivardecode PRIVATE -fsanitize=address,undefined)
  if(TARGET api_test)
    target_link_options(api_test PRIVATE -fsanitize=address,undefined)
    set_target_properties(api_test PROPERTIES LINKER_LANGUAGE CXX)
  endif()
  add_executable(evd_fuzz tests/fuzz.cpp)
  target_compile_features(evd_fuzz PRIVATE cxx_std_17)
  target_compile_definitions(evd_fuzz PRIVATE EVD_TEST_REGISTRY="${CMAKE_CURRENT_SOURCE_DIR}/registry")
  target_compile_options(evd_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(evd_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(evd_fuzz PRIVATE efivardecode_lib)
endif()
